// ===== Save report as PDF (single page, clean header) ===== window.saveReportAsPDF = function() { const card = document.querySelector('#tab-report .card'); if (!card) return; // استنساخ البطاقة لتعديلها const clone = card.cloneNode(true); // إزالة رأس البطاقة القديم (العنوان والأزرار) const oldHead = clone.querySelector('.card-head'); if (oldHead) oldHead.remove(); // إنشاء رأس جديد للتقرير const { name } = getAuth(); const userName = name ? name.split(' ')[0] : 'Guest'; const now = new Date(); const printDate = now.toLocaleDateString() + ' ' + now.toLocaleTimeString(); const newHeader = document.createElement('div'); newHeader.className = 'report-header'; newHeader.innerHTML = `

Health Report

Patient: ${escapeHtml(userName)} Printed: ${escapeHtml(printDate)}
`; clone.prepend(newHeader); // خيارات html2pdf لصفحة A4 واحدة const opt = { margin: [0.5, 0.5, 0.5, 0.5], // top, right, bottom, left (inches) filename: `Health_Report_${new Date().toISOString().slice(0,10)}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, letterRendering: true, useCORS: true, logging: false }, jsPDF: { unit: 'in', format: 'a4', orientation: 'portrait' }, pagebreak: { mode: ['css', 'legacy'] } // لتجنب التقسيم غير المرغوب }; // توليد PDF وحفظه html2pdf().set(opt).from(clone).save(); }; // ===== Share report as PDF ===== window.shareReport = function() { const card = document.querySelector('#tab-report .card'); if (!card) return; const clone = card.cloneNode(true); const oldHead = clone.querySelector('.card-head'); if (oldHead) oldHead.remove(); const { name } = getAuth(); const userName = name ? name.split(' ')[0] : 'Guest'; const now = new Date(); const printDate = now.toLocaleDateString() + ' ' + now.toLocaleTimeString(); const newHeader = document.createElement('div'); newHeader.className = 'report-header'; newHeader.innerHTML = `

Health Report

Patient: ${escapeHtml(userName)} Printed: ${escapeHtml(printDate)}
`; clone.prepend(newHeader); const opt = { margin: [0.5, 0.5, 0.5, 0.5], filename: `Health_Report_${new Date().toISOString().slice(0,10)}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, letterRendering: true, useCORS: true, logging: false }, jsPDF: { unit: 'in', format: 'a4', orientation: 'portrait' }, pagebreak: { mode: ['css', 'legacy'] } }; html2pdf().from(clone).set(opt).toPdf().get('pdf').then(function(pdf) { const blob = pdf.output('blob'); const file = new File([blob], opt.filename, { type: 'application/pdf' }); if (navigator.share && navigator.canShare && navigator.canShare({ files: [file] })) { navigator.share({ title: 'Health Report', files: [file] }).catch(err => { toast('Sharing failed: ' + err.message, 'err'); pdf.save(opt.filename); // fallback }); } else { pdf.save(opt.filename); toast('Sharing not supported, file saved instead', 'ok'); } }); };
